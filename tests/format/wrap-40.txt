// A narrow layout expands parameter, field, array and nested argument lists.
struct Measurements {
    first_measurement: i64,
    second_measurement: i64,
    third_measurement: i64
}

fn combine(
    first_measurement: i64,
    second_measurement: i64,
    third_measurement: i64
) -> i64 {
    first_measurement +
        second_measurement +
        third_measurement
}

fn main() {
    let first_measurement = 11
    let second_measurement = 22
    let third_measurement = 33
    let data = Measurements {
        first_measurement,
        second_measurement,
        third_measurement
    }
    let results = [
        combine(
            first_measurement,
            second_measurement,
            third_measurement
        ),
        combine(4, 5, 6),
        combine(7, 8, 9)
    ]
    println(results[0] +
        data.third_measurement)
}

fn checked(
    first_value: i64,
    second_value: i64
) -> Result((), i64) {
    if first_value < 2 { return Err(1) }
    Ok(())
}

fn describe(
    text: str,
    code: i64
) -> String {
    text.to_string()
}

fn report(
    code: i64
) -> Result(i64, String) {
    if code > 1 {
        return Err(describe(
            "not accepted",
            code
        ))
    }
    let shown = Some(describe(
        "accepted",
        code
    ).as_str().len)
    Ok(code)
}

// Long expressions break after their loosest operators, except where a
// block on the line is what is long.
fn limits(
    first_value: i64,
    second_value: i64
) -> bool {
    if first_value < 2 &&
        second_value > 3 {
        return false
    }
    if first_value > 100 ||
        second_value > 200 ||
        first_value == second_value {
        return true
    }
    first_value * 2 + second_value * 3 >
        10
}

// A long one-line block or match opens onto lines of its own, and the
// blocks of an `if ... else` chain open together.
fn choose(value: Option(i64)) -> i64 {
    let picked = match value {
        Some(v) => v * 2,
        None => 0
    }
    if picked > 10 {
        picked - 10
    } else if picked > 5 {
        picked - 5
    } else {
        0
    }
}
